package com.danismp.auction;

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

import com.danismp.auction.gui.ActiveAuctionScreen;
import com.danismp.auction.gui.ConfigScreen;
import com.danismp.auction.gui.ResultScreen;

import net.minecraft.client.MinecraftClient;
import net.minecraft.item.ItemStack;

/**
 * Holds everything about the auction that is currently running (or the one that just ended).
 *
 * <p>This lives outside the screens, so payments keep being detected and the countdown keeps
 * running while no auction GUI is open.
 */
public final class AuctionManager {
	public enum Phase {
		/** No auction has been started yet. */
		IDLE,
		/** Counting down and accepting payments. */
		RUNNING,
		/** Countdown frozen, payments still accepted. */
		PAUSED,
		/** Ended naturally or was stopped by the streamer. */
		FINISHED
	}

	/** Immutable snapshot used by the results screen and the main menu summary. */
	public record AuctionResult(ItemStack itemStack, String winnerName, long winningAmount,
			long totalReceived, int paymentCount, boolean stoppedManually) {
		public boolean hasWinner() {
			return winnerName != null;
		}
	}

	private static final AuctionManager INSTANCE = new AuctionManager();

	public static AuctionManager get() {
		return INSTANCE;
	}

	private Phase phase = Phase.IDLE;
	private ItemStack itemStack = ItemStack.EMPTY;
	private long durationMillis;
	private long remainingMillis;
	private long deadlineMillis;
	private long minimumPayment;
	private final List<Payment> payments = new ArrayList<>();
	private Payment highest;
	private long totalReceived;
	private int nextOrder = 1;
	private AuctionResult lastResult;

	/** Guards against the same system line being processed twice. */
	private String lastMessageText;
	private long lastMessageAt;

	private AuctionManager() {
	}

	// ---------------------------------------------------------------- queries

	public Phase phase() {
		return phase;
	}

	public boolean isActive() {
		return phase == Phase.RUNNING || phase == Phase.PAUSED;
	}

	public boolean isPaused() {
		return phase == Phase.PAUSED;
	}

	public ItemStack itemStack() {
		return itemStack;
	}

	public long minimumPayment() {
		return minimumPayment;
	}

	public long durationMillis() {
		return durationMillis;
	}

	public List<Payment> payments() {
		return Collections.unmodifiableList(payments);
	}

	public int paymentCount() {
		return payments.size();
	}

	public Payment highestPayment() {
		return highest;
	}

	public long totalReceived() {
		return totalReceived;
	}

	public AuctionResult lastResult() {
		return lastResult;
	}

	/** Milliseconds left on the clock, live for a running auction and frozen while paused. */
	public long remainingMillis() {
		if (phase == Phase.RUNNING) {
			return Math.max(0L, deadlineMillis - System.currentTimeMillis());
		}
		if (phase == Phase.PAUSED) {
			return remainingMillis;
		}
		return 0L;
	}

	/** Whole seconds left, rounded up so the clock only reads 00:00 once the auction is over. */
	public long remainingSeconds() {
		long millis = remainingMillis();
		return millis <= 0 ? 0 : (millis + 999L) / 1000L;
	}

	// ---------------------------------------------------------------- lifecycle

	/**
	 * Starts a fresh auction. All previous payment data is discarded here, so nothing carries
	 * over into the new auction.
	 */
	public void start(ItemStack stack, long durationSeconds, long minimumPayment) {
		this.itemStack = stack.copy();
		this.durationMillis = durationSeconds * 1000L;
		this.remainingMillis = durationMillis;
		this.deadlineMillis = System.currentTimeMillis() + durationMillis;
		this.minimumPayment = minimumPayment;
		this.payments.clear();
		this.highest = null;
		this.totalReceived = 0L;
		this.nextOrder = 1;
		this.phase = Phase.RUNNING;
		this.lastResult = null;
		this.lastMessageText = null;

		MinecraftClient client = MinecraftClient.getInstance();
		client.setScreen(new ActiveAuctionScreen());
	}

	public void pause() {
		if (phase == Phase.RUNNING) {
			remainingMillis = Math.max(0L, deadlineMillis - System.currentTimeMillis());
			phase = Phase.PAUSED;
		}
	}

	public void resume() {
		if (phase == Phase.PAUSED) {
			deadlineMillis = System.currentTimeMillis() + remainingMillis;
			phase = Phase.RUNNING;
		}
	}

	public void togglePause() {
		if (phase == Phase.RUNNING) {
			pause();
		} else if (phase == Phase.PAUSED) {
			resume();
		}
	}

	/**
	 * Streamer pressed "Stop Auction": the auction ends immediately, no further payments are
	 * accepted and the configuration menu opens again. The outcome is kept as a manual stop so
	 * it can be told apart from a normal finish.
	 */
	public void stopManually() {
		if (!isActive()) {
			return;
		}

		phase = Phase.FINISHED;
		remainingMillis = 0L;
		lastResult = snapshot(true);
		MinecraftClient.getInstance().setScreen(new ConfigScreen());
	}

	/** Called every client tick while an auction is running. */
	public void tick() {
		if (phase == Phase.RUNNING && System.currentTimeMillis() >= deadlineMillis) {
			completeNaturally();
		}
	}

	private void completeNaturally() {
		phase = Phase.FINISHED;
		remainingMillis = 0L;
		lastResult = snapshot(false);
		MinecraftClient.getInstance().setScreen(new ResultScreen(lastResult));
	}

	private AuctionResult snapshot(boolean stoppedManually) {
		Payment top = highest;
		return new AuctionResult(
				itemStack.copy(),
				top == null ? null : top.player(),
				top == null ? 0L : top.amount(),
				totalReceived,
				payments.size(),
				stoppedManually);
	}

	// ---------------------------------------------------------------- payments

	/**
	 * Entry point for system chat lines. Only exact economy messages for the running (or paused)
	 * auction are accepted, and only payments reaching the configured minimum.
	 */
	public void onSystemMessage(String rawLine) {
		if (rawLine == null || rawLine.isBlank()) {
			return;
		}

		PaymentParser.Detected detected = PaymentParser.parse(rawLine);
		if (detected == null) {
			if (looksLikeEconomyLine(rawLine)) {
				DaniAuctionClient.LOGGER.info("Auction: could not read economy line: {}", rawLine);
			}
			return;
		}

		if (!isActive()) {
			DaniAuctionClient.LOGGER.info("Auction: payment of {} from {} ignored, no auction running",
					detected.amount(), detected.player());
			return;
		}

		long now = System.currentTimeMillis();
		if (rawLine.equals(lastMessageText) && now - lastMessageAt < 60L) {
			// The very same line arriving twice in the same moment is a duplicate delivery,
			// not a second payment.
			return;
		}
		lastMessageText = rawLine;
		lastMessageAt = now;

		if (detected.amount() < minimumPayment) {
			DaniAuctionClient.LOGGER.debug("Auction: payment of {} from {} is below the minimum {}",
					detected.amount(), detected.player(), minimumPayment);
			return;
		}

		accept(detected.player(), detected.amount());
	}

	private static boolean looksLikeEconomyLine(String line) {
		return line.startsWith("ECONOMY") || line.contains("Dollar zeichen") || line.contains(" erhalten!");
	}

	/** Registers one individual payment. Payments are never merged per player. */
	public void accept(String player, long amount) {
		Payment payment = new Payment(player, amount, nextOrder++);
		payments.add(payment);
		totalReceived = saturatingAdd(totalReceived, amount);

		if (highest == null || amount > highest.amount()) {
			highest = payment;
		}
	}

	private static long saturatingAdd(long a, long b) {
		long sum = a + b;
		return sum < 0 && a > 0 && b > 0 ? Long.MAX_VALUE : sum;
	}
}

/**
 * Colours of the auction interface: a dark translucent panel with a neon gradient frame that runs
 * from the first colour on the left to the second on the right, plus one highlight colour.
 * Everything is configurable in the appearance menu and persisted as JSON.
 */
public final class AuctionTheme {
	public static final AuctionTheme INSTANCE = new AuctionTheme();

	private static final int DEFAULT_GRADIENT_START = 0x3ADFD4;
	private static final int DEFAULT_GRADIENT_END = 0xC24BEA;
	private static final int DEFAULT_ACCENT = 0x63E6E6;
	private static final int DEFAULT_BACKDROP = 0x000000;
	private static final int DEFAULT_OPACITY = 40;

	private static final Gson GSON = new GsonBuilder().setPrettyPrinting().create();

	/** Left end of the neon frame gradient. */
	private int gradientTop = DEFAULT_GRADIENT_START;
	/** Right end of the neon frame gradient. */
	private int gradientBottom = DEFAULT_GRADIENT_END;
	private int accent = DEFAULT_ACCENT;
	private int backdrop = DEFAULT_BACKDROP;
	private int opacity = DEFAULT_OPACITY;

	private AuctionTheme() {
	}

	// ------------------------------------------------------------------ colours

	public int gradientTop() {
		return gradientTop;
	}

	public int gradientBottom() {
		return gradientBottom;
	}

	public int accent() {
		return accent;
	}

	public int backdrop() {
		return backdrop;
	}

	public int opacity() {
		return opacity;
	}

	public void setGradientTop(int rgb) {
		this.gradientTop = rgb & 0xFFFFFF;
	}

	public void setGradientBottom(int rgb) {
		this.gradientBottom = rgb & 0xFFFFFF;
	}

	public void setAccent(int rgb) {
		this.accent = rgb & 0xFFFFFF;
	}

	public void setBackdrop(int rgb) {
		this.backdrop = rgb & 0xFFFFFF;
	}

	public void setOpacity(int percent) {
		this.opacity = Math.max(10, Math.min(100, percent));
	}

	public void reset() {
		this.gradientTop = DEFAULT_GRADIENT_START;
		this.gradientBottom = DEFAULT_GRADIENT_END;
		this.accent = DEFAULT_ACCENT;
		this.backdrop = DEFAULT_BACKDROP;
		this.opacity = DEFAULT_OPACITY;
	}

	// ------------------------------------------------------------------ derived colours

	/** Dark panel alpha; low enough to keep the world visible, high enough to read on. */
	public float panelAlpha() {
		return 0.42f + 0.48f * (opacity / 100f);
	}

	public int panelTop() {
		return argb(0x0A0913, panelAlpha());
	}

	public int panelBottom() {
		return argb(0x05050B, Math.min(0.98f, panelAlpha() + 0.12f));
	}

	public int cardTop() {
		return argb(0x121021, Math.min(0.98f, panelAlpha() + 0.02f));
	}

	public int cardBottom() {
		return argb(0x0A0916, Math.min(0.98f, panelAlpha() + 0.10f));
	}

	public int backdropTop() {
		return argb(backdrop, 0.32f);
	}

	public int backdropBottom() {
		return argb(backdrop, 0.44f);
	}

	public static int argb(int rgb, float alpha) {
		int a = Math.max(0, Math.min(255, Math.round(255 * alpha)));
		return (a << 24) | (rgb & 0xFFFFFF);
	}

	public static int darken(int rgb, float factor) {
		int r = Math.round(((rgb >> 16) & 0xFF) * factor);
		int g = Math.round(((rgb >> 8) & 0xFF) * factor);
		int b = Math.round((rgb & 0xFF) * factor);
		return (r << 16) | (g << 8) | b;
	}

	public static int lighten(int rgb, float factor) {
		int r = Math.min(255, Math.round(((rgb >> 16) & 0xFF) + (255 - ((rgb >> 16) & 0xFF)) * factor));
		int g = Math.min(255, Math.round(((rgb >> 8) & 0xFF) + (255 - ((rgb >> 8) & 0xFF)) * factor));
		int b = Math.min(255, Math.round((rgb & 0xFF) + (255 - (rgb & 0xFF)) * factor));
		return (r << 16) | (g << 8) | b;
	}

	// ------------------------------------------------------------------ persistence

	private static Path file() {
		return FabricLoader.getInstance().getConfigDir().resolve("danismp-auction-theme.json");
	}

	public void load() {
		Path path = file();
		if (!Files.exists(path)) {
			return;
		}

		try {
			String json = Files.readString(path, StandardCharsets.UTF_8);
			JsonObject root = JsonParser.parseString(json).getAsJsonObject();

			if (root.has("gradientTop")) {
				this.gradientTop = parseHex(root.get("gradientTop").getAsString(), DEFAULT_GRADIENT_START);
			}
			if (root.has("gradientBottom")) {
				this.gradientBottom = parseHex(root.get("gradientBottom").getAsString(), DEFAULT_GRADIENT_END);
			}
			if (root.has("accent")) {
				this.accent = parseHex(root.get("accent").getAsString(), DEFAULT_ACCENT);
			}
			if (root.has("backdrop")) {
				this.backdrop = parseHex(root.get("backdrop").getAsString(), DEFAULT_BACKDROP);
			}
			if (root.has("opacity")) {
				setOpacity(root.get("opacity").getAsInt());
			}
		} catch (Exception e) {
			DaniAuctionClient.LOGGER.warn("Could not read the auction theme, using defaults", e);
		}
	}

	public void save() {
		JsonObject root = new JsonObject();
		root.addProperty("gradientTop", hex(gradientTop));
		root.addProperty("gradientBottom", hex(gradientBottom));
		root.addProperty("accent", hex(accent));
		root.addProperty("backdrop", hex(backdrop));
		root.addProperty("opacity", opacity);

		try {
			Path path = file();
			Files.createDirectories(path.getParent());
			Files.writeString(path, GSON.toJson(root), StandardCharsets.UTF_8);
		} catch (IOException e) {
			DaniAuctionClient.LOGGER.warn("Could not save the auction theme", e);
		}
	}

	public static String hex(int rgb) {
		return String.format("#%06X", rgb & 0xFFFFFF);
	}

	public static int parseHex(String value, int fallback) {
		if (value == null) {
			return fallback;
		}

		String clean = value.trim();
		if (clean.startsWith("#")) {
			clean = clean.substring(1);
		}
		if (clean.length() != 6) {
			return fallback;
		}

		try {
			return Integer.parseInt(clean, 16) & 0xFFFFFF;
		} catch (NumberFormatException e) {
			return fallback;
		}
	}
}

			if (overlay) {
				return;
			}

			AuctionManager.get().onSystemMessage(message.getString());
		});

		LOGGER.info("DaniSMP Auction ready - press K to open the auction menu");
	}
}

package com.danismp.auction;

import java.util.Locale;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

/**
 * Parses the DaniSMP economy system message and formats money values.
 *
 * <p>The live server format, matched as a whole chat line, is:
 * <pre>ECONOMY ➥ Du hast $10K von RyvanX erhalten!</pre>
 * The sender and the amount are captured independently, so any player name and any amount work:
 * <pre>ECONOMY ➥ Du hast 10M von Fred123 erhalten!</pre>
 *
 * <p>The separator glyph between {@code ECONOMY} and {@code Du hast} is not fixed, and the
 * currency symbol may be any short punctuation run, so the exact presentation of the plugin does
 * not matter. Everything else has to match exactly.
 *
 * <p>The older documented layout ({@code ECONOMY> Du hast (user) (amount) Dollar zeichen
 * gesendet!}) is still recognised.
 *
 * <p>{@link Matcher#matches()} is used on purpose: a line has to match in full, so player chatter
 * that merely mentions the words, or a system line with extra text before or after the message,
 * is rejected.
 */
public final class PaymentParser {
	/** {@code ECONOMY ➥ Du hast $10K von RyvanX erhalten!} */
	private static final Pattern RECEIVED_PATTERN = Pattern.compile(
			"\\[?ECONOMY\\]?\\s*[^\\p{L}\\p{N}$]{0,8}\\s*Du hast\\s+[^\\p{L}\\p{N}\\s]{0,3}\\s*"
					+ "([0-9][0-9.,]*)\\s*([A-Za-z]{0,8})\\s+von\\s+([^!]{1,40}?)\\s+erhalten!",
			Pattern.CASE_INSENSITIVE | Pattern.UNICODE_CASE);

	/** {@code ECONOMY> Du hast RyvanX 1000000 Dollar zeichen gesendet!} */
	private static final Pattern LEGACY_PATTERN = Pattern.compile(
			"\\[?ECONOMY\\]?\\s*[^\\p{L}\\p{N}$]{0,8}\\s*Du hast\\s+([^!]{1,40}?)\\s+"
					+ "([0-9][0-9.,]*)\\s*([A-Za-z]{0,8})\\s+Dollar zeichen gesendet!",
			Pattern.CASE_INSENSITIVE | Pattern.UNICODE_CASE);

	/** 18 digits stay inside the signed long range, so absurd amounts are rejected instead of overflowing. */
	private static final int MAX_AMOUNT_DIGITS = 18;

	private PaymentParser() {
	}

	/** A successfully parsed payment. */
	public record Detected(String player, long amount) {
	}

	/**
	 * @param rawLine the plain text of a system chat line
	 * @return the parsed payment, or {@code null} when the line is not a valid economy message
	 */
	public static Detected parse(String rawLine) {
		if (rawLine == null || rawLine.isBlank()) {
			return null;
		}

		String line = rawLine.replace('\u00A0', ' ').strip();

		Matcher received = RECEIVED_PATTERN.matcher(line);
		if (received.matches()) {
			String player = received.group(3).strip();
			long amount = parseAmount(received.group(1), received.group(2));
			return amount > 0L && !player.isEmpty() ? new Detected(player, amount) : null;
		}

		Matcher legacy = LEGACY_PATTERN.matcher(line);
		if (legacy.matches()) {
			String player = legacy.group(1).strip();
			long amount = parseAmount(legacy.group(2), legacy.group(3));
			return amount > 0L && !player.isEmpty() ? new Detected(player, amount) : null;
		}

		return null;
	}

	/**
	 * Reads an amount such as {@code 10}, {@code 10K}, {@code 1.5M}, {@code 10m}, {@code 3,200,000}
	 * or {@code 1.000.000} together with its optional suffix.
	 *
	 * @return the amount in whole dollars, or {@code -1} when it cannot be read
	 */
	private static long parseAmount(String numberToken, String suffixToken) {
		String digits = numberToken == null ? "" : numberToken.trim();
		while (!digits.isEmpty() && (digits.endsWith(".") || digits.endsWith(","))) {
			digits = digits.substring(0, digits.length() - 1);
		}
		if (digits.isEmpty()) {
			return -1L;
		}

		long multiplier = suffixMultiplier(suffixToken == null ? "" : suffixToken.trim());
		if (multiplier <= 0L) {
			return -1L;
		}

		int lastSeparator = Math.max(digits.lastIndexOf('.'), digits.lastIndexOf(','));
		int trailingDigits = digits.length() - lastSeparator - 1;
		int separators = 0;
		for (int i = 0; i < digits.length(); i++) {
			char c = digits.charAt(i);
			if (c == '.' || c == ',') {
				separators++;
			}
		}

		// For an abbreviated amount (10K, 1.5M, 1.234Mio) a single separator is a decimal point.
		// Amounts written out in full (3,200,000 / 1.000.000) keep separators as thousands marks.
		boolean decimalAbbreviation = multiplier > 1L && separators == 1 && lastSeparator >= 0
				&& trailingDigits >= 1 && trailingDigits <= 3;

		double value;
		try {
			if (decimalAbbreviation) {
				// Abbreviated amount with a decimal fraction, e.g. 1.5M / 2,5M.
				String whole = digits.substring(0, lastSeparator).replace(".", "").replace(",", "");
				String fraction = digits.substring(lastSeparator + 1);
				if (!whole.matches("\\d{1,15}") || !fraction.matches("\\d{1,3}")) {
					return -1L;
				}
				value = Double.parseDouble(whole + "." + fraction);
			} else {
				String plain = digits.replace(".", "").replace(",", "");
				if (!plain.matches("\\d{1," + MAX_AMOUNT_DIGITS + "}")) {
					return -1L;
				}
				value = Double.parseDouble(plain);
			}
		} catch (NumberFormatException e) {
			return -1L;
		}

		double total = value * multiplier;
		if (!(total > 0) || total >= 9.0e18) {
			return -1L;
		}
		return Math.round(total);
	}

	/** Multiplier for the amount suffix; {@code -1} marks an unknown suffix. */
	private static long suffixMultiplier(String suffix) {
		if (suffix.isEmpty()) {
			return 1L;
		}

		return switch (suffix.toUpperCase(Locale.ROOT)) {
			case "K", "TSND", "TAUSEND" -> 1_000L;
			case "M", "MIO", "MIL", "MILLION", "MILLIONEN" -> 1_000_000L;
			case "B", "MRD", "MILLIARDE", "MILLIARDEN", "BILLION" -> 1_000_000_000L;
			case "T", "BIO" -> 1_000_000_000_000L;
			default -> -1L;
		};
	}

	/** Formats money the way the auction UI shows it, e.g. {@code $5,000,000}. */
	public static String money(long amount) {
		return "$" + String.format(Locale.ROOT, "%,d", amount);
	}
}

	@Override
	protected void build() {
		int x = contentX();
		int w = contentWidth();
		int top = contentTop();
		int bottom = toolbarY() - 10;

		int statH = compact ? 40 : 50;
		int receivedW = Math.round(w * 0.40f);
		int timerWanted = Math.round(w * 0.32f);
		int countWanted = w - receivedW - timerWanted - GAP * 2;

		statsX = x;
		statsY = top;
		statsW = receivedW;
		statsH = statH;
		timerX = x + receivedW + GAP;
		timerY = top;
		this.timerW = timerWanted;
		timerH = statH;
		countX = timerX + timerWanted + GAP;
		countY = top;
		this.countW = countWanted;
		countH = statH;

		int bodyTop = top + statH + GAP;
		int bodyH = Math.max(60, bottom - bodyTop);

		if (compact) {
			int half = (w - GAP) / 2;
			itemX = x;
			itemY = bodyTop;
			itemW = half;
			itemH = Math.min(72, Math.max(54, bodyH / 2));
			topX = x + half + GAP;
			topY = bodyTop;
			topW = w - half - GAP;
			topH = itemH;
			listX = x;
			listY = bodyTop + itemH + GAP;
			listW = w;
			listH = Math.max(30, bottom - listY);
		} else {
			int leftW = Math.round(w * 0.40f);
			itemX = x;
			itemY = bodyTop;
			itemW = leftW;
			itemH = Math.max(112, Math.min(158, Math.round(bodyH * 0.48f)));
			topX = x;
			topY = itemY + itemH + GAP;
			topW = leftW;
			topH = Math.max(54, bodyTop + bodyH - topY);
			listX = x + leftW + GAP;
			listY = bodyTop;
			listW = w - leftW - GAP;
			listH = bodyH;
		}

		pauseButton = addButtonRow(x, toolbarY(), w, BUTTON_H,
				new String[] { "PAUSE", "STOP", "CONFIG", "THEME", "CLOSE" },
				new DarkButton.Style[] { DarkButton.Style.SECONDARY, DarkButton.Style.DANGER,
						DarkButton.Style.GHOST, DarkButton.Style.GHOST, DarkButton.Style.GHOST },
				new Runnable[] { manager::togglePause, manager::stopManually,
						() -> this.client.setScreen(new ConfigScreen()),
						() -> this.client.setScreen(new ThemeSettingsScreen(this)), this::close },

	@Override
	protected void drawContent(DrawContext context, int mouseX, int mouseY, float delta) {
		this.chips.clear();

		boolean paused = manager.isPaused();
		if (pauseButton != null) {
			String wanted = paused ? "RESUME" : "PAUSE";
			if (!pauseButton.getMessage().getString().equals(wanted)) {
				pauseButton.setMessage(Ui.literal(wanted));
			}
		}

		drawHeader(context, "DaniSMP AuctionMod", null, paused ? "PAUSED" : "LIVE",
				paused ? Ui.AMBER : Ui.GREEN);
		drawSection(context, paused ? "AUCTION PAUSED" : "AUCTION LIVE",
				paused ? "Timer stopped - payments still count" : "Every chat payment becomes a bid");

		if (!manager.isActive()) {
			drawEmptyState(context);
			drawToolbar(context);
			return;
		}

		trackNewPayments();
		ItemStack stack = manager.itemStack();
		itemLabel = ItemLabel.of(stack);
		itemDetail = ItemLabel.describeLines(stack);

		drawStatCard(context, statsX, statsY, statsW, statsH, "TOTAL RECEIVED",
				PaymentParser.money(manager.totalReceived()), Ui.accent());
		drawTimerCard(context);
		drawStatCard(context, countX, countY, countW, countH, "PAYMENTS",
				String.valueOf(manager.paymentCount()), Ui.TEXT);

		drawItemCard(context);
		drawHighestCard(context, mouseX, mouseY);
		drawPaymentList(context, mouseX, mouseY, delta);
		drawToolbar(context);
	}

	private void drawEmptyState(DrawContext context) {
		int centerX = panelX + panelW / 2;
		int centerY = panelY + panelH / 2 - 12;
		Ui.textCentered(context, this.textRenderer, "NO AUCTION RUNNING", centerX, centerY, Ui.TEXT_MUTED);
		Ui.textCentered(context, this.textRenderer, "Press CONFIG to set one up.", centerX, centerY + 14,
				Ui.TEXT_DIM);
	}

	private void drawTimerCard(DrawContext context) {
		Ui.card(context, timerX, timerY, timerW, timerH);
		boolean paused = manager.isPaused();
		long remaining = manager.remainingSeconds();
		int color = paused ? Ui.AMBER : (remaining <= 10 ? Ui.RED : Ui.TEXT);

		Ui.text(context, this.textRenderer, "TIME LEFT", timerX + 10, timerY + 8, Ui.TEXT_DIM);

		float scale = compact ? 1.28f : 1.55f;
		String clock = Ui.clock(remaining);
		String shown = Ui.fitText(this.textRenderer, clock, (int) ((timerW - 34) / scale));
		Ui.textScaled(context, this.textRenderer, shown, timerX + 10,
				timerY + timerH - Math.round(9 * scale) - 8, scale, color);

		if (paused) {
			Ui.roundedRect(context, timerX + timerW - 17, timerY + 11, 7, 7, 3, Ui.AMBER);
		} else {
			drawPulseDot(context, timerX + timerW - 18, timerY + 11, 8, Ui.GREEN);
		}
	}

	private void drawItemCard(DrawContext context) {
		Ui.card(context, itemX, itemY, itemW, itemH);
		ItemStack stack = manager.itemStack();
		int ix = itemX + 10;
		Ui.text(context, this.textRenderer, "AUCTIONED ITEM", ix, itemY + 8, Ui.TEXT_DIM);

		boolean dense = itemH >= 108 && !compact;
		int iconSize = compact ? 20 : 32;
		float iconScale = iconSize / 16f;
		int iconX = ix;
		int iconY = itemY + (compact ? 22 : 26);
		itemIconX = iconX;
		itemIconY = iconY;
		itemIconSize = iconSize;

		if (stack.isEmpty()) {
			Ui.roundedRect(context, iconX - 4, iconY - 4, iconSize + 8, iconSize + 8, 5, Ui.border(0.2f));
			Ui.textCentered(context, this.textRenderer, "?", iconX + iconSize / 2, iconY + iconSize / 2 - 4,
					Ui.TEXT_DIM);
		} else {
			drawItemBadge(context, stack, iconX, iconY, iconScale);
		}

		int textX = iconX + iconSize + 12;
		int textRight = itemX + itemW - 10;
		int textW = Math.max(14, textRight - textX);

		int chipW = Ui.tagChip(context, this.textRenderer, textX, iconY - (compact ? 0 : 1),
				itemLabel.prefix(), itemLabel.prefixColor());
		int nameY = chipW > 0 ? iconY + 16 : iconY + 2;
		Ui.text(context, this.textRenderer, Ui.fit(this.textRenderer, itemLabel.name(), textW), textX, nameY,
				Ui.TEXT);

		drawDetailLines(context, textX, nameY + 13, textW, itemY + itemH - (itemH >= 108 ? 48 : 8));

		if (itemH >= 108) {
			int fy = itemY + itemH - 42;
			Ui.divider(context, ix, fy - 8, itemW - 20);
			Ui.text(context, this.textRenderer, "MIN BID", ix, fy, Ui.TEXT_DIM);
			Ui.textRight(context, this.textRenderer, PaymentParser.money(manager.minimumPayment()), textRight,
					fy, Ui.accent());
			Ui.text(context, this.textRenderer, "DURATION", ix, fy + 13, Ui.TEXT_DIM);
			Ui.textRight(context, this.textRenderer, manager.durationMillis() / 1000 + "s", textRight,
					fy + 13, Ui.TEXT_MUTED);
		}
	}

	/** Every enchantment and lore line the card has room for. */
	private void drawDetailLines(DrawContext context, int x, int y, int width, int limit) {
		if (itemDetail.isEmpty()) {
			if (y + 9 <= limit) {
				Ui.text(context, this.textRenderer, "No enchantments", x, y, Ui.TEXT_DIM);
			}
			return;
		}

		int cursor = y;
		for (int i = 0; i < itemDetail.size(); i++) {
			if (cursor + 11 > limit) {
				if (cursor + 9 <= limit) {
					Ui.text(context, this.textRenderer, "+" + (itemDetail.size() - i) + " more", x, cursor,
							Ui.TEXT_DIM);
				}
				return;
			}

			Ui.text(context, this.textRenderer,
					Ui.fitText(this.textRenderer, itemDetail.get(i), width), x, cursor, Ui.TEXT_DIM);
			cursor += 11;
		}
	}

	private void drawHighestCard(DrawContext context, int mouseX, int mouseY) {
		Payment top = manager.highestPayment();

		if (top == null) {
			Ui.card(context, topX, topY, topW, topH);
			Ui.text(context, this.textRenderer, "HIGHEST BID", topX + 10, topY + 8, Ui.TEXT_DIM);
			Ui.text(context, this.textRenderer,
					Ui.fitText(this.textRenderer, "Waiting for the first payment", topW - 20),
					topX + 10, topY + 24, Ui.TEXT_MUTED);
			return;
		}

		Ui.accentCard(context, topX, topY, topW, topH, 1f);

		int x = topX + 10;
		int right = topX + topW - 10;
		Ui.text(context, this.textRenderer, "HIGHEST BID", x, topY + 7, Ui.TEXT_MUTED);

		int head = Math.min(compact ? 18 : 24, Math.max(12, topH - 32));
		int headY = topY + (compact ? 18 : 20);
		Ui.head(context, this.textRenderer, x, headY, head, top.player());

		int textX = x + head + 8;
		int textW = Math.max(20, right - textX);
		float nameScale = compact ? 1f : 1.08f;
		Ui.textScaled(context, this.textRenderer,
				Ui.fitText(this.textRenderer, top.player(), (int) (textW / nameScale)),
				textX, headY + 1, nameScale, 0xFFFFFFFF);
		Ui.text(context, this.textRenderer,
				Ui.fitText(this.textRenderer, PaymentParser.money(top.amount()), textW),
				textX, headY + Math.round(9 * nameScale) + 1, Ui.HIGHLIGHT);

		int chipsWidth = this.chips.width(this.textRenderer);
		if (topH >= 76 && topW >= chipsWidth + 20) {
			this.chips.draw(context, this.textRenderer, x, topY + topH - 19, top.player(), mouseX, mouseY);
		}
	}

	private void drawPaymentList(DrawContext context, int mouseX, int mouseY, float delta) {
		Ui.card(context, listX, listY, listW, listH);

		int innerX = listX + 9;
		int innerW = listW - 18;

		Ui.text(context, this.textRenderer, "PAYMENTS", innerX, listY + 8, Ui.TEXT_DIM);
		String countLabel = manager.paymentCount()
				+ (manager.paymentCount() == 1 ? " payment" : " payments");
		Ui.textRight(context, this.textRenderer, countLabel, listX + listW - 10, listY + 8, Ui.TEXT_MUTED);
		Ui.divider(context, innerX, listY + 21, innerW);

		int rowsTop = listY + 25;
		int rowsBottom = listY + listH - 7;
		int rowH = compact ? 20 : 24;

		List<Payment> payments = manager.payments();
		int total = payments.size();

		if (total == 0) {
			int centerY = (rowsTop + rowsBottom) / 2 - 8;
			Ui.textCentered(context, this.textRenderer, "WAITING FOR PAYMENTS", innerX + innerW / 2,
					centerY, Ui.TEXT_MUTED);
			if (!compact) {
				Ui.textCentered(context, this.textRenderer, "Accepted chat payments appear here.",
						innerX + innerW / 2, centerY + 13, Ui.TEXT_DIM);
			}
			return;
		}

		if (rowsBottom - rowsTop < rowH) {
			return;
		}

		int visible = Math.max(1, (rowsBottom - rowsTop) / rowH);
		scrollRows = Math.max(0, Math.min(Math.max(0, total - visible), scrollRows));
		scrollAnim += (scrollRows - scrollAnim) * Math.min(1f, Math.max(delta, 0.1f) * 0.45f);

		int start = Math.max(0, total - visible - Math.round(scrollAnim));
		int count = Math.min(visible, total - start);

		boolean showChips = !compact && innerW >= 300;
		int chipsWidth = showChips ? this.chips.width(this.textRenderer) : 0;

		Payment highest = manager.highestPayment();
		float glow = 1f - Math.min(1f, (System.currentTimeMillis() - newestPaymentAt) / 1600f);
		boolean mouseInside = mouseX >= innerX && mouseX < innerX + innerW
				&& mouseY >= rowsTop && mouseY < rowsBottom;

		context.enableScissor(innerX, rowsTop, innerX + innerW, rowsBottom);

		for (int i = 0; i < count; i++) {
			int index = start + i;
			Payment payment = payments.get(index);
			int rowH2 = rowH - 2;
			int rowY = rowsBottom - (count - i) * rowH;

			boolean hovered = mouseInside && mouseY >= rowY && mouseY < rowY + rowH2;
			boolean isHighest = highest != null && payment.order() == highest.order();
			boolean isNewest = index == total - 1;

			int background = isHighest ? Ui.accent(0.34f)
					: (index % 2 == 0 ? 0x4D000000 : 0x26000000);
			if (hovered) {
				background = Ui.lerp(background, Ui.accent(0.28f), 0.7f);
			}
			if (isNewest && glow > 0f) {
				background = Ui.lerp(background, Ui.accent(0.38f), glow);
			}
			Ui.roundedRect(context, innerX, rowY, innerW, rowH2, 4, background);
			if (isHighest) {
				Ui.roundedRect(context, innerX, rowY, 2, rowH2, 1, Ui.accent(0.95f));
			}

			int centerY = rowY + rowH2 / 2;

			int badgeSize = compact ? 12 : 14;
			int badgeX = innerX + 5;
			Ui.badge(context, badgeX, centerY - badgeSize / 2, badgeSize,
					isHighest ? (0xFF000000 | Ui.accent()) : 0x66000000,
					isHighest ? Ui.accentLight() : Ui.border(0.3f));
			Ui.textCentered(context, this.textRenderer, String.valueOf(payment.order()),
					badgeX + badgeSize / 2, centerY - 4,
					isHighest ? Ui.TEXT_ON_ACCENT : Ui.TEXT_MUTED);

			int headSize = compact ? 14 : 16;
			int headX = badgeX + badgeSize + 5;
			Ui.head(context, this.textRenderer, headX, centerY - headSize / 2, headSize, payment.player());

			int nameX = headX + headSize + 7;
			String amount = PaymentParser.money(payment.amount());
			int amountWidth = Ui.width(this.textRenderer, amount);
			int chipsX = innerX + innerW - 6 - chipsWidth;
			int amountX = showChips ? chipsX - 8 - amountWidth : innerX + innerW - 8 - amountWidth;
			int nameMax = Math.max(10, amountX - nameX - 6);

			Ui.text(context, this.textRenderer,
					Ui.fitText(this.textRenderer, payment.player(), nameMax), nameX, centerY - 4,
					isHighest ? Ui.HIGHLIGHT : Ui.TEXT);
			Ui.text(context, this.textRenderer, amount, amountX, centerY - 4,
					isHighest ? Ui.accentLight() : Ui.TEXT_MUTED);

			if (showChips) {
				this.chips.draw(context, this.textRenderer, chipsX, centerY - 6, payment.player(), mouseX,
						mouseY);
			}
		}

		context.disableScissor();
		drawScrollbar(context, innerX, innerW, rowsTop, rowsBottom, total, visible);
	}

	private void drawScrollbar(DrawContext context, int innerX, int innerW, int rowsTop, int rowsBottom,
			int total, int visible) {
		int max = Math.max(0, total - visible);
		if (max <= 0) {
			return;
		}

		int trackX = innerX + innerW - 3;
		int trackH = rowsBottom - rowsTop;
		Ui.rect(context, trackX, rowsTop, 2, trackH, 0x33FFFFFF);

		int thumbH = Math.max(10, Math.round(trackH * (visible / (float) total)));
		float progress = 1f - (Math.round(scrollAnim) / (float) max);
		int thumbY = rowsTop + Math.round((trackH - thumbH) * progress);
		Ui.rect(context, trackX, thumbY, 2, thumbH, Ui.accent(0.85f));
	}

	private void trackNewPayments() {
		int total = manager.paymentCount();
		if (lastSeenPayments < 0) {
			lastSeenPayments = total;
			return;
		}

		if (total > lastSeenPayments) {
			int added = total - lastSeenPayments;
			lastSeenPayments = total;
			newestPaymentAt = System.currentTimeMillis();
			if (scrollRows > 0) {
				scrollRows += added;
			}
		}
	}

	// ------------------------------------------------------------------ input

	@Override
	public boolean mouseClicked(Click click, boolean doubled) {
		if (click.button() == 0 && this.chips.click(click.x(), click.y())) {
			return true;
		}
		return super.mouseClicked(click, doubled);
	}

	@Override
	public boolean mouseScrolled(double mouseX, double mouseY, double horizontalAmount, double verticalAmount) {
		boolean insideList = mouseX >= listX && mouseX < listX + listW
				&& mouseY >= listY + 25 && mouseY < listY + listH;

		int rowH = compact ? 20 : 24;
		int visible = Math.max(1, (listY + listH - 7 - (listY + 25)) / rowH);
		int max = Math.max(0, manager.paymentCount() - visible);

		if (insideList && max > 0) {
			scrollRows = Math.max(0, Math.min(max, scrollRows + (int) Math.signum(verticalAmount) * 2));
			return true;
		}

		return super.mouseScrolled(mouseX, mouseY, horizontalAmount, verticalAmount);
	}

	@Override
	protected void drawOverlay(DrawContext context, int mouseX, int mouseY, float delta) {
		ItemStack stack = manager.itemStack();
		if (stack.isEmpty() || itemIconSize <= 0) {
			return;
		}

		if (mouseX >= itemIconX && mouseX < itemIconX + itemIconSize
				&& mouseY >= itemIconY && mouseY < itemIconY + itemIconSize) {
			context.drawItemTooltip(this.textRenderer, stack, mouseX, mouseY);
		}
	}
}

/**
 * Shared chrome for every auction screen: dark neon panel, centred gradient title, optional tab
 * bar, centred section heading and a row of wide bars at the bottom.
 */
public abstract class AuctionScreen extends Screen {
	protected static final int MARGIN = 12;
	protected static final int BUTTON_H = 24;
	/** Creator credit, painted tiny into the screen corner. */
	private static final String WATERMARK = "by ryvan7";

	protected int panelX;
	protected int panelY;
	protected int panelW;
	protected int panelH;
	protected int pad = 16;
	protected boolean compact;

	private final List<FieldBox> fieldBoxes = new ArrayList<>();
	private long openedAt;

	protected AuctionScreen(Text title) {
		super(title);
	}

	@Override
	protected void init() {
		this.compact = this.height < 360 || this.width < 520;
		this.pad = compact ? 12 : 16;

		this.panelW = Math.min(maxPanelWidth(), this.width - MARGIN * 2);
		this.panelH = Math.min(maxPanelHeight(), this.height - MARGIN * 2);
		this.panelX = (this.width - panelW) / 2;
		this.panelY = (this.height - panelH) / 2;

		if (openedAt == 0L) {
			openedAt = System.currentTimeMillis();
		}

		Ui.clearSkinCache();
		this.fieldBoxes.clear();
		build();
	}

	protected int maxPanelWidth() {
		return 620;
	}

	protected int maxPanelHeight() {
		return 430;
	}

	protected abstract void build();

	protected abstract void drawContent(DrawContext context, int mouseX, int mouseY, float delta);

	protected void drawOverlay(DrawContext context, int mouseX, int mouseY, float delta) {
	}

	// ------------------------------------------------------------------ geometry

	protected int contentX() {
		return panelX + pad;
	}

	protected int contentWidth() {
		return panelW - pad * 2;
	}

	protected int titleY() {
		return panelY + (compact ? 8 : 10);
	}

	protected int sectionTop() {
		return titleY() + (compact ? 22 : 26);
	}

	protected int contentTop() {
		return sectionTop() + (compact ? 17 : 22);
	}

	protected int toolbarY() {
		return panelY + panelH - pad - BUTTON_H;
	}

	protected float openProgress() {
		return Ui.easeOut(Math.min(1f, (System.currentTimeMillis() - openedAt) / 240f));
	}

	protected float pulse(long periodMillis) {
		double phase = (System.currentTimeMillis() % periodMillis) / (double) periodMillis;
		return (float) (0.5 + 0.5 * Math.sin(phase * Math.PI * 2));
	}

	@Override
	public boolean shouldPause() {
		return false;
	}

	// ------------------------------------------------------------------ rendering

	@Override
	public void renderBackground(DrawContext context, int mouseX, int mouseY, float delta) {
		Ui.backdrop(context, this.width, this.height);
		Ui.window(context, panelX, panelY, panelW, panelH);
		drawContent(context, mouseX, mouseY, delta);
	}

	@Override
	public void render(DrawContext context, int mouseX, int mouseY, float delta) {
		super.render(context, mouseX, mouseY, delta);
		drawWatermark(context);
		drawOverlay(context, mouseX, mouseY, delta);
	}

	/** Very small credit in the top-right corner of the screen, clear of every panel. */
	private void drawWatermark(DrawContext context) {
		float scale = 0.5f;
		int x = this.width - 4 - Math.round(this.textRenderer.getWidth(WATERMARK) * scale);
		Ui.textScaled(context, this.textRenderer, WATERMARK, x, 4, scale,
				Ui.withAlpha(Ui.TEXT_DIM, 0.55f));
	}

	protected void drawToolbar(DrawContext context) {
		Ui.accentRule(context, contentX(), toolbarY() - 10, contentWidth());
	}

	/** Colour-gradient title with an optional status chip on the right. */
	protected void drawHeader(DrawContext context, String title, String subtitle, String statusLabel,
			int statusColor) {
		float scale = compact ? 1.25f : 1.5f;
		int unscaled = Math.max(1, this.textRenderer.getWidth(title));
		int maxWidth = contentWidth() - (statusLabel == null ? 0 : 62);
		if (unscaled * scale > maxWidth) {
			scale = Math.max(0.8f, maxWidth / (float) unscaled);
		}
		Ui.gradientTitle(context, this.textRenderer, title, panelX + panelW / 2, titleY(), scale,
				Ui.gradientLeft(), Ui.gradientRight());

		if (statusLabel != null) {
			int chipWidth = Ui.width(this.textRenderer, statusLabel) + 14;
			Ui.chip(context, this.textRenderer, panelX + panelW - pad - chipWidth, titleY() + 1, statusLabel,
					statusColor, Ui.withAlpha(statusColor, 0.14f));

	/** Lays a row of wide bars across {@code w}. */
	protected DarkButton[] addButtonRow(int x, int y, int w, int h, String[] labels,
			DarkButton.Style[] styles, Runnable[] actions, float[] weights) {
		float total = 0f;
		for (float weight : weights) {
			total += weight;
		}

		int gap = 6;
		int available = w - gap * (labels.length - 1);
		int cursor = x;
		DarkButton[] buttons = new DarkButton[labels.length];

		for (int i = 0; i < labels.length; i++) {
			int buttonWidth = i == labels.length - 1
					? x + w - cursor
					: Math.max(46, Math.round(available * weights[i] / total));
			buttons[i] = addButton(cursor, y, buttonWidth, h, labels[i], styles[i], actions[i]);
			cursor += buttonWidth + gap;
		}
		return buttons;
	}

	protected TextFieldWidget addNumberField(int x, int y, int width, int height, String placeholder,
			String initial) {
		// Vanilla only centres field text vertically when it paints its own background, so the widget
		// covers the text line itself and the themed box is drawn around it.
		int textHeight = this.textRenderer.fontHeight;
		int textY = y + (height - textHeight) / 2;
		TextFieldWidget field = new TextFieldWidget(this.textRenderer, x, textY, width, textHeight,
				Ui.literal(placeholder));
		field.setMaxLength(18);
		field.setTextPredicate(value -> value.isEmpty() || value.matches("\\d{1,18}"));
		field.setDrawsBackground(false);
		field.setCentered(true);
		field.setEditableColor(Ui.TEXT);
		field.setUneditableColor(Ui.TEXT_DIM);
		field.setPlaceholder(Ui.literal(placeholder).formatted(Formatting.DARK_GRAY));
		field.setText(initial == null ? "" : initial);
		addDrawableChild(field);
		this.fieldBoxes.add(new FieldBox(field, x, y, width, height));
		return field;
	}

	protected void drawField(DrawContext context, TextFieldWidget field, boolean focused) {
		if (field == null) {
			return;
		}

		FieldBox box = boxFor(field);
		int x = box != null ? box.x() : field.getX();
		int y = box != null ? box.y() : field.getY();
		int w = box != null ? box.w() : field.getWidth();
		int h = box != null ? box.h() : field.getHeight();

		context.fillGradient(x, y, x + w, y + h, 0xE60C0A16, 0xE607060E);
		if (focused) {
			context.fillGradient(x + 2, y + 2, x + w - 2, y + h - 2,
					Ui.withAlpha(Ui.gradientLeft(), 0.14f), Ui.withAlpha(Ui.gradientRight(), 0.14f));
		}
		Ui.neonFrame(context, x, y, w, h, focused ? 2 : 1, focused ? 1f : 0.45f, focused);
	}

	private FieldBox boxFor(TextFieldWidget field) {
		for (FieldBox box : this.fieldBoxes) {
			if (box.field() == field) {
				return box;
			}
		}
		return null;
	}

	/** Clicking anywhere in the visible box focuses the field, not just its thin text line. */
	@Override
	public boolean mouseClicked(Click click, boolean doubled) {
		for (FieldBox box : this.fieldBoxes) {
			if (click.button() == 0 && click.x() >= box.x() && click.x() < box.x() + box.w()
					&& click.y() >= box.y() && click.y() < box.y() + box.h()) {
				this.setFocused(box.field());
				box.field().mouseClicked(click, doubled);
				return true;
			}
		}
		return super.mouseClicked(click, doubled);
	}

	private record FieldBox(TextFieldWidget field, int x, int y, int w, int h) {
	}

	protected static long parseLong(String value, long fallback) {
		if (value == null) {
			return fallback;
		}

		String digits = value.replace(".", "").replace(",", "").replace(" ", "").trim();
		if (digits.isEmpty() || digits.length() > 18) {
			return fallback;
		}

		try {
			return Long.parseLong(digits);
		} catch (NumberFormatException e) {
			return fallback;
		}
	}
}

package com.danismp.auction.gui;

import net.minecraft.client.MinecraftClient;
import net.minecraft.client.font.TextRenderer;

import net.minecraft.client.gui.widget.ClickableWidget;
import net.minecraft.text.Text;

/**
 * Wide dark bar with a neon gradient frame, centred vanilla-font label and an eased hover state.
 */
public class DarkButton extends ClickableWidget {
	public enum Style {
		/** Main call to action. */
		PRIMARY,
		/** Solid highlight colour, used for winning bid actions. */
		GOLD,
		/** Normal option bar. */
		SECONDARY,
		/** Destructive action. */
		DANGER,
		/** Low emphasis. */
		GHOST,
		/** Selected tab. */
		TAB_ACTIVE,
		/** Inactive tab. */
		TAB_INACTIVE
	}

	private final Style style;
	private final Runnable action;
	private float hover;
	private float press;

	public DarkButton(int x, int y, int width, int height, Text message, Style style, Runnable action) {
		super(x, y, width, height, message);
		this.style = style;
		this.action = action;
	}

	public DarkButton withTooltip(Text tooltip) {
		setTooltip(Tooltip.of(tooltip));
		return this;
	}

	@Override
	protected void renderWidget(DrawContext context, int mouseX, int mouseY, float delta) {
		float step = Math.min(1f, Math.max(delta, 0.1f) * 0.5f);
		hover += ((isHovered() && this.active ? 1f : 0f) - hover) * step;
		press = Math.max(0f, press - step * 2f);

		int x = getX();
		int y = getY() + (press > 0.5f ? 1 : 0);
		int w = getWidth();
		int h = getHeight();
		float alpha = this.active ? 1f : 0.4f;
		TextRenderer tr = MinecraftClient.getInstance().textRenderer;

		if (this.style == Style.TAB_ACTIVE || this.style == Style.TAB_INACTIVE) {
			Ui.tab(context, tr, x, y, w, h, getMessage().getString(), this.style == Style.TAB_ACTIVE, hover);
			return;
		}

		int fillTop;
		int fillBottom;
		float frameAlpha;
		int textColor = Ui.TEXT;
		boolean glow = false;

		switch (this.style) {
			case PRIMARY -> {
				fillTop = Ui.lerp(0xE6151230, 0xFF232052, hover);
				fillBottom = Ui.lerp(0xE60C0A1A, 0xFF141033, hover);
				frameAlpha = 1f;
				glow = true;
			}
			case GOLD -> {
				fillTop = Ui.lerp(0xE6301B44, 0xFF44265E, hover);
				fillBottom = Ui.lerp(0xE61A0E26, 0xFF2A1440, hover);
				frameAlpha = 0.95f;
				glow = true;
			}
			case SECONDARY -> {
				fillTop = Ui.lerp(0xE6121020, 0xFF1D1936, hover);
				fillBottom = Ui.lerp(0xE60B0914, 0xFF120F26, hover);
				frameAlpha = 0.55f + hover * 0.35f;
			}
			case DANGER -> {
				fillTop = Ui.lerp(0xE6250F1E, 0xFF3A1730, hover);
				fillBottom = Ui.lerp(0xE6160812, 0xFF250A1C, hover);
				frameAlpha = 0.8f + hover * 0.2f;
				textColor = 0xFFFFC7D8;
			}
			default -> {
				fillTop = Ui.lerp(0xC00E0C18, 0xF01A1730, hover);
				fillBottom = Ui.lerp(0xC0080710, 0xF0100D22, hover);
				frameAlpha = 0.28f + hover * 0.4f;
				textColor = hover > 0.4f ? Ui.TEXT : Ui.TEXT_MUTED;
			}
		}

		context.fillGradient(x, y, x + w, y + h, Ui.withAlpha(fillTop, alpha),
				Ui.withAlpha(fillBottom, alpha));
		context.fillGradient(x + 2, y + 2, x + w - 2, y + Math.max(8, h / 2), 0x18FFFFFF, 0x00FFFFFF);
		Ui.neonFrame(context, x, y, w, h, 2, frameAlpha * alpha, glow);

		if (isFocused()) {
			Ui.outline(context, x - 2, y - 2, w + 4, h + 4, Ui.withAlpha(Ui.gradient(0.5f), 0.55f));
		}

		Text label = Ui.font(getMessage());
		context.drawTextWithShadow(tr, label, x + (w - tr.getWidth(label)) / 2, y + (h - 9) / 2,
				Ui.withAlpha(textColor, alpha));
	}

	@Override
	public void onClick(Click click, boolean doubled) {
		if (!this.active) {
			return;
		}

		this.press = 1f;
		playClickSound(MinecraftClient.getInstance().getSoundManager());
		this.action.run();
	}

	@Override
	protected void appendClickableNarrations(NarrationMessageBuilder builder) {
		appendDefaultNarrations(builder);
	}

}

import net.minecraft.component.DataComponentTypes;
import net.minecraft.component.type.ItemEnchantmentsComponent;
import net.minecraft.enchantment.Enchantment;
import net.minecraft.enchantment.EnchantmentHelper;
import net.minecraft.item.ItemStack;
import net.minecraft.registry.entry.RegistryEntry;
import net.minecraft.text.MutableText;
import net.minecraft.text.Text;

/**
 * Splits an item's display name into its bracket rank/tag prefix and the name itself, keeping the
 * server's formatting (colour, bold) intact.
 */
public final class ItemLabel {
	private static final Pattern BRACKET_PREFIX = Pattern.compile("^\\s*(\\[[^\\]]{1,24}\\])\\s*(.+)$");

	private final String prefix;
	private final Text name;
	private final String plainName;

	private ItemLabel(String prefix, Text name, String plainName) {
		this.prefix = prefix;
		this.name = name;
		this.plainName = plainName;
	}

	public static ItemLabel of(ItemStack stack) {
		if (stack == null || stack.isEmpty()) {
			return new ItemLabel(null, Text.literal("Unknown item"), "Unknown item");
		}

		Text full = stack.getName();
		String plain = full.getString().replace('\u00A0', ' ').trim();

		Matcher matcher = BRACKET_PREFIX.matcher(plain);
		if (matcher.matches()) {
			MutableText rest = Text.literal(matcher.group(2).trim());
			if (full.getStyle().getColor() != null || !full.getStyle().isEmpty()) {
				rest.setStyle(full.getStyle());
			}
			return new ItemLabel(matcher.group(1), rest, rest.getString());
		}

		return new ItemLabel(null, full, plain);
	}

	/** Bracket loot tag from the name, or {@code null} when the name has none. */
	public String prefix() {
		return prefix;
	}

	/** Display name with its original formatting preserved. */
	public Text name() {
		return name;
	}

	public String plainName() {
		return plainName;
	}

	/**
	 * Only the item's enchantments, one per line. Lore, kit notes and attribute text from the
	 * tooltip are deliberately left out, so the card shows nothing but "Sharpness V" and friends.
	 */
	public static List<String> describeLines(ItemStack stack) {
		List<String> lines = new ArrayList<>();
		if (stack == null || stack.isEmpty()) {
			return lines;
		}

		ItemEnchantmentsComponent stored = stack.get(DataComponentTypes.STORED_ENCHANTMENTS);
		ItemEnchantmentsComponent enchantments = stored != null && !stored.isEmpty()
				? stored
				: EnchantmentHelper.getEnchantments(stack);

		for (RegistryEntry<Enchantment> enchantment : enchantments.getEnchantments()) {
			lines.add(Enchantment.getName(enchantment, enchantments.getLevel(enchantment)).getString());
		}
		return lines;
	}

	/** Colours the tag chip: the accent for a bracket loot tag. */
	public int prefixColor() {
		return prefix == null ? Ui.TEXT_DIM : Ui.accent();
	}
}

package com.danismp.auction.gui;

import java.util.ArrayList;
import java.util.List;

import net.minecraft.client.MinecraftClient;
import net.minecraft.client.font.TextRenderer;
import net.minecraft.client.gui.DrawContext;

/**
 * The TPA / HERE / COPY chips that sit next to a player name.
 */
public final class PlayerChips {
	private static final int CHIP_HEIGHT = 13;

	public enum Action {
		TPA, HERE, COPY
	}

	private record Hit(int x, int y, int w, int h, String player, Action action) {
	}

	private final List<Hit> hits = new ArrayList<>();

	public void clear() {
		this.hits.clear();
	}

	public int width(TextRenderer tr) {
		return Ui.width(tr, "TPA") + Ui.width(tr, "HERE") + Ui.width(tr, "COPY") + 3 * 12 + 6;
	}

	public void draw(DrawContext context, TextRenderer tr, int x, int y, String player, int mouseX, int mouseY) {
		int cursor = x;
		cursor += chip(context, tr, cursor, y, "TPA", Action.TPA, player, mouseX, mouseY) + 3;
		cursor += chip(context, tr, cursor, y, "HERE", Action.HERE, player, mouseX, mouseY) + 3;
		chip(context, tr, cursor, y, "COPY", Action.COPY, player, mouseX, mouseY);
	}

	private int chip(DrawContext context, TextRenderer tr, int x, int y, String label, Action action,
			String player, int mouseX, int mouseY) {
		int w = Ui.width(tr, label) + 12;
		boolean hovered = mouseX >= x && mouseX < x + w && mouseY >= y && mouseY < y + CHIP_HEIGHT;
		Ui.smallChip(context, tr, x, y, label, hovered);
		this.hits.add(new Hit(x, y, w, CHIP_HEIGHT, player, action));
		return w;
	}

	/** Runs the action of the chip under the cursor. */
	public boolean click(double mouseX, double mouseY) {
		for (Hit hit : this.hits) {
			if (mouseX >= hit.x() && mouseX < hit.x() + hit.w()
					&& mouseY >= hit.y() && mouseY < hit.y() + hit.h()) {
				perform(hit.player(), hit.action());
				return true;
			}
		}
		return false;
	}

	private void perform(String player, Action action) {
		MinecraftClient client = MinecraftClient.getInstance();
		switch (action) {
			case TPA -> command(client, "tpa " + player);
			case HERE -> command(client, "tphere " + player);
			case COPY -> client.keyboard.setClipboard(player);
		}
	}

	private void command(MinecraftClient client, String command) {
		if (client.getNetworkHandler() != null) {
			client.getNetworkHandler().sendChatCommand(command);
		}
	}
}

	// ------------------------------------------------------------------ hsv helpers

	private static void toHsv(int rgb, float[] out) {
		float r = ((rgb >> 16) & 0xFF) / 255f;
		float g = ((rgb >> 8) & 0xFF) / 255f;
		float b = (rgb & 0xFF) / 255f;

		float max = Math.max(r, Math.max(g, b));
		float min = Math.min(r, Math.min(g, b));
		float delta = max - min;

		float hue = 0f;
		if (delta > 0.0001f) {
			if (max == r) {
				hue = 60f * (((g - b) / delta) % 6f);
			} else if (max == g) {
				hue = 60f * ((b - r) / delta + 2f);
			} else {
				hue = 60f * ((r - g) / delta + 4f);
			}
		}
		if (hue < 0f) {
			hue += 360f;
		}

		out[0] = hue;
		out[1] = max <= 0f ? 0f : delta / max;
		out[2] = max;
	}

	private static int fromHsv(float[] hsv) {
		float hue = hsv[0] % 360f;
		float saturation = Ui.clamp01(hsv[1]);
		float value = Ui.clamp01(hsv[2]);

		float c = value * saturation;
		float x = c * (1f - Math.abs((hue / 60f) % 2f - 1f));
		float m = value - c;

		float r;
		float g;
		float b;
		switch ((int) Math.floor(hue / 60f) % 6) {
			case 0 -> {
				r = c;
				g = x;
				b = 0f;
			}
			case 1 -> {
				r = x;
				g = c;
				b = 0f;
			}
			case 2 -> {
				r = 0f;
				g = c;
				b = x;
			}
			case 3 -> {
				r = 0f;
				g = x;
				b = c;
			}
			case 4 -> {
				r = x;
				g = 0f;
				b = c;
			}
			default -> {
				r = c;
				g = 0f;
				b = x;
			}
		}

		int ri = Math.round((r + m) * 255f);
		int gi = Math.round((g + m) * 255f);
		int bi = Math.round((b + m) * 255f);
		return (ri << 16) | (gi << 8) | bi;
	}

package com.danismp.auction.gui;

import java.util.function.IntConsumer;

import net.minecraft.client.gui.Click;

import net.minecraft.client.gui.widget.ClickableWidget;

/**
 * Labelled slider used by the appearance menu. Drag the track or click anywhere on it.
 */
public class ThemeSlider extends ClickableWidget {
	private static final int LABEL_WIDTH = 12;
	private static final int VALUE_WIDTH = 34;

	private final String label;
	private final int min;
	private final int max;
	private final IntConsumer onChange;

	private int value;
	private boolean dragging;
	private float hover;

	public ThemeSlider(int x, int y, int width, int height, String label, int min, int max, int value,
			IntConsumer onChange) {
		super(x, y, width, height, Text.literal(label));
		this.label = label;
		this.min = min;
		this.max = max;
		this.onChange = onChange;
		this.value = Math.max(min, Math.min(max, value));
	}

	public int value() {
		return value;
	}

	public void setValue(int newValue) {
		int clamped = Math.max(min, Math.min(max, newValue));
		if (clamped != value) {
			value = clamped;
			onChange.accept(clamped);
		}
	}

	@Override
	protected void renderWidget(DrawContext context, int mouseX, int mouseY, float delta) {
		float step = Math.min(1f, Math.max(delta, 0.1f) * 0.5f);
		hover += ((isHovered() || dragging ? 1f : 0f) - hover) * step;

		int x = getX();
		int y = getY();
		int w = getWidth();
		int h = getHeight();

		int trackX = x + LABEL_WIDTH;
		int trackW = Math.max(12, w - LABEL_WIDTH - VALUE_WIDTH);
		int trackY = y + h / 2 - 2;
		float progress = (value - min) / (float) Math.max(1, max - min);

		float textY = y + (h - 9) / 2f;
		Ui.text(context, this.textRenderer(), label, x, Math.round(textY), Ui.TEXT_DIM);

		Ui.roundedRect(context, trackX, trackY, trackW, 4, 2, 0x77000000);
		int filled = Math.round(trackW * progress);
		if (filled > 0) {
			Ui.roundedRect(context, trackX, trackY, Math.max(3, filled), 4, 2, Ui.accent(0.9f));
		}

		int knobX = trackX + filled;
		Ui.roundedRect(context, knobX - 3, trackY - 2, 7, 8, 3, 0xCC000000);
		Ui.roundedRect(context, knobX - 2, trackY - 1, 5, 6, 2,
				hover > 0.4f ? 0xFFFFFFFF : (isFocused() ? Ui.accentLight() : Ui.TEXT_MUTED));

		Ui.textRight(context, this.textRenderer(), String.valueOf(value), x + w, Math.round(textY),
				hover > 0.4f ? Ui.TEXT : Ui.TEXT_MUTED);

		if (isFocused()) {
			Ui.outline(context, x - 2, y, w + 2, h, Ui.accent(0.35f));
		}
	}

	private net.minecraft.client.font.TextRenderer textRenderer() {
		return net.minecraft.client.MinecraftClient.getInstance().textRenderer;
	}

	@Override
	public void onClick(Click click, boolean doubled) {
		setFocused(true);
		this.dragging = true;
		setFromMouse(click.x());
	}

	@Override
	protected void onDrag(Click click, double offsetX, double offsetY) {
		this.dragging = true;
		setFromMouse(click.x());
	}

	@Override
	public void onRelease(Click click) {
		this.dragging = false;
	}

	@Override
	public boolean mouseDragged(Click click, double offsetX, double offsetY) {
		if (!this.dragging) {
			return false;
		}
		setFromMouse(click.x());
		return true;
	}

	@Override
	public boolean mouseReleased(Click click) {
		boolean wasDragging = this.dragging;
		this.dragging = false;
		return wasDragging;
	}

	private void setFromMouse(double mouseX) {
		int trackX = getX() + LABEL_WIDTH;
		int trackW = Math.max(12, getWidth() - LABEL_WIDTH - VALUE_WIDTH);
		float t = (float) ((mouseX - trackX) / (double) trackW);
		setValue(Math.round(min + Ui.clamp01(t) * (max - min)));
	}

	@Override
	protected void appendClickableNarrations(NarrationMessageBuilder builder) {
		appendDefaultNarrations(builder);
	}
}

	public static int lerp(int from, int to, float t) {
		t = clamp01(t);
		int a = (int) (((from >>> 24) & 0xFF) + (((to >>> 24) & 0xFF) - ((from >>> 24) & 0xFF)) * t);
		int r = (int) (((from >> 16) & 0xFF) + (((to >> 16) & 0xFF) - ((from >> 16) & 0xFF)) * t);
		int g = (int) (((from >> 8) & 0xFF) + (((to >> 8) & 0xFF) - ((from >> 8) & 0xFF)) * t);
		int b = (int) ((from & 0xFF) + ((to & 0xFF) - (from & 0xFF)) * t);
		return (a << 24) | (r << 16) | (g << 8) | b;
	}

	public static float clamp01(float v) {
		return v < 0f ? 0f : (v > 1f ? 1f : v);

	public static float easeOut(float t) {
		t = clamp01(t);
		return 1f - (1f - t) * (1f - t);
	}

	public static void roundedRect(DrawContext ctx, int x, int y, int w, int h, int radius, int color) {
		if (w <= 0 || h <= 0) {
			return;
		}

		int r = Math.min(radius, Math.min(w / 2, h / 2));
		if (r <= 0) {
			rect(ctx, x, y, w, h, color);
			return;
		}

		for (int i = 0; i < r; i++) {
			double dy = r - i;
			int inset = r - (int) Math.round(Math.sqrt(Math.max(0, (double) r * r - dy * dy)));
			rect(ctx, x + inset, y + i, w - inset * 2, 1, color);
			rect(ctx, x + inset, y + h - 1 - i, w - inset * 2, 1, color);
		}
		rect(ctx, x, y + r, w, h - r * 2, color);
	}

	 * second on the right, exactly like the reference design.

			boolean glow) {
		if (w <= 0 || h <= 0) {
			return;
		}

		int t = Math.max(1, Math.min(thickness, Math.min(w / 2, h / 2)));
		int left = withAlpha(gradientLeft(), alpha);
		int right = withAlpha(gradientRight(), alpha);

		if (glow) {
			ctx.fillGradient(x - 1, y - 1, x + w + 1, y + t + 3, withAlpha(gradientLeft(), alpha * 0.22f),

		}

		ctx.fillGradient(x, y, x + w, y + t, left, right);
		ctx.fillGradient(x, y + h - t, x + w, y + h, left, right);
		ctx.fill(x, y + t, x + t, y + h - t, left);
		ctx.fill(x + w - t, y + t, x + w, y + h - t, right);
	}

		ctx.fillGradient(x + 2, y + 2, x + w - 2, y + h - 2, accent(0.20f * alpha), 0x00FFFFFF);
		neonFrame(ctx, x, y, w, h, 2, 0.95f * alpha, true);
	}

	public static void pill(DrawContext ctx, int x, int y, int w, int h, int fill, int border) {
		int r = Math.max(2, h / 2);
		roundedRect(ctx, x, y, w, h, r, border);
		roundedRect(ctx, x + 1, y + 1, w - 2, h - 2, Math.max(1, r - 1), fill);
	}

	}

	/** Tab bar item from the reference design: dark fill with a neon frame when active. */
	public static void tab(DrawContext ctx, TextRenderer tr, int x, int y, int w, int h, String label,
			boolean active, float hover) {
		if (active) {
			ctx.fillGradient(x, y, x + w, y + h, 0xE61B1740, 0xE6100E2A);
			ctx.fillGradient(x + 2, y + 2, x + w - 2, y + h - 2, withAlpha(gradientLeft(), 0.26f),
					withAlpha(gradientRight(), 0.26f));
		} else {
			ctx.fillGradient(x, y, x + w, y + h, 0xE60C0B14, 0xE607060D);
			if (hover > 0.01f) {
				ctx.fillGradient(x + 2, y + 2, x + w - 2, y + h - 2, 0x18FFFFFF, 0x00FFFFFF);
			}
		}

		neonFrame(ctx, x, y, w, h, 2, active ? 1f : 0.30f + hover * 0.35f, active);
		textCentered(ctx, tr, label, x + w / 2, y + (h - 9) / 2, active ? 0xFFFFFFFF : TEXT_DIM);
	}

	public static void textScaled(DrawContext ctx, TextRenderer tr, Text value, int x, int y, float scale,
			int color) {
		ctx.getMatrices().pushMatrix();
		ctx.getMatrices().translate(x, y);
		ctx.getMatrices().scale(scale, scale);
		ctx.drawTextWithShadow(tr, value, 0, 0, color);
		ctx.getMatrices().popMatrix();
	}

	/** Big centred title painted with the neon gradient, like the reference header. */
	public static void gradientTitle(DrawContext ctx, TextRenderer tr, String value, int centerX, int y,
			float scale, int left, int right) {
		int unscaled = Math.max(1, tr.getWidth(value));
		int startX = centerX - Math.round(unscaled * scale) / 2;

		ctx.getMatrices().pushMatrix();
		ctx.getMatrices().translate(startX, y);
		ctx.getMatrices().scale(scale, scale);

		int cursor = 0;
		for (int i = 0; i < value.length(); i++) {
			String ch = String.valueOf(value.charAt(i));
			int color = lerp(left, right, cursor / (float) unscaled);
			ctx.drawTextWithShadow(tr, literal(ch), cursor, 0, color);
			cursor += tr.getWidth(ch);
		}

		ctx.getMatrices().popMatrix();
	}

	/** Truncates until the text plus an ellipsis fits, never exceeding {@code maxWidth}. */
	public static String fitText(TextRenderer tr, String value, int maxWidth) {
		if (value == null || value.isEmpty() || maxWidth <= 0) {
			return "";
		}
		if (width(tr, value) <= maxWidth) {
			return value;
		}

		String dots = "...";
		int dotsWidth = width(tr, dots);
		if (dotsWidth > maxWidth) {
			return "";
		}

		StringBuilder sb = new StringBuilder();
		for (int i = 0; i < value.length(); i++) {
			char c = value.charAt(i);
			if (width(tr, sb.toString() + c) + dotsWidth > maxWidth) {
				break;
			}
			sb.append(c);
		}
		return sb.isEmpty() ? "" : sb + dots;
	}

	/** Same as {@link #fitText} but keeps the colour and per-part formatting of a component. */
	public static Text fit(TextRenderer tr, Text value, int maxWidth) {
		if (value == null || maxWidth <= 0) {
			return Text.empty();
		}
		if (tr.getWidth(value) <= maxWidth) {
			return value;
		}

		StringVisitable trimmed = tr.trimToWidth(value, Math.max(1, maxWidth - width(tr, "...")));
		Text base = trimmed instanceof Text text ? text : Text.literal(trimmed.getString());
		MutableText result = base.copy();
		result.append(literal("..."));
		return result;
	}

	// ------------------------------------------------------------------ heads

	public static void head(DrawContext ctx, TextRenderer tr, int x, int y, int size, String name) {
		if (name == null || name.isEmpty()) {
			name = "?";
		}

		Identifier skin = skinFor(name);
		if (skin != null) {
			ctx.getMatrices().pushMatrix();
			ctx.getMatrices().translate(x, y);
			float scale = size / 8f;
			ctx.getMatrices().scale(scale, scale);
			ctx.drawTexture(RenderPipelines.GUI_TEXTURED, skin, 0, 0, 8, 8, 8, 8, 64, 64);
			ctx.drawTexture(RenderPipelines.GUI_TEXTURED, skin, 0, 0, 40, 8, 8, 8, 64, 64);
			ctx.getMatrices().popMatrix();
		} else {
			avatar(ctx, tr, x, y, size, name);
		}

		outline(ctx, x, y, size, size, 0x99000000);
	}

	private static Identifier skinFor(String name) {
		String key = name.toLowerCase(Locale.ROOT);
		Identifier cached = SKIN_CACHE.get(key);
		if (cached != null) {
			return cached;
		}

		MinecraftClient client = MinecraftClient.getInstance();
		if (client.getNetworkHandler() == null) {
			return null;
		}

		PlayerListEntry entry = client.getNetworkHandler().getPlayerListEntry(name);
		if (entry == null) {
			entry = client.getNetworkHandler().getCaseInsensitivePlayerInfo(name);
		}
		if (entry == null) {
			return null;
		}

		SkinTextures skin = entry.getSkinTextures();
		if (skin == null || skin.body() == null) {
			return null;
		}

		Identifier texture = skin.body().texturePath();
		if (texture == null) {
			return null;
		}

		if (SKIN_CACHE.size() > 128) {
			SKIN_CACHE.clear();
		}
		SKIN_CACHE.put(key, texture);
		return texture;
	}

	public static void avatar(DrawContext ctx, TextRenderer tr, int x, int y, int size, String name) {
		int color = AVATAR_COLORS[Math.floorMod(name == null ? 0 : name.hashCode(), AVATAR_COLORS.length)];
		roundedRect(ctx, x, y, size, size, Math.max(2, size / 4), color);
		ctx.fillGradient(x, y, x + size, y + size, 0x22FFFFFF, 0x33000000);

		String initial = name == null || name.isEmpty() ? "?" : name.substring(0, 1).toUpperCase(Locale.ROOT);
		float scale = Math.max(0.75f, size / 12f);
		int textY = y + Math.round((size - 9 * scale) / 2f);
		textScaledCentered(ctx, tr, initial, x + size / 2, textY, scale, 0xFFF2F8F0);
	}

	public static String clock(long totalSeconds) {
		long seconds = Math.max(0L, totalSeconds);
		long hours = seconds / 3600L;
		long minutes = (seconds % 3600L) / 60L;
		long secs = seconds % 60L;
		if (hours > 0) {
			return String.format(Locale.ROOT, "%d:%02d:%02d", hours, minutes, secs);
		}
		return String.format(Locale.ROOT, "%02d:%02d", minutes, secs);
	}
